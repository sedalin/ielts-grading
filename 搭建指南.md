# IELTS 作文智能批改项目 - 完整搭建指南

## 项目概述

这是一个基于 AI 的雅思作文批改系统，使用 MiniMax API 进行智能评分，提供：
- 4 个维度的评分：Task Response、Coherence and Cohesion、Lexical Resource、Grammatical Range and Accuracy
- 交互式错误高亮显示
- 详细的改进建议表格
- 错误类型过滤切换

async function callMiniMax(messages: MiniMaxMessage[]): Promise<string> {
  const response = await fetch(`${MINIMAX_BASE_URL}/text/chatcompletion_v2`, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'Authorization': `Bearer ${MINIMAX_API_KEY}`,
    },
    body: JSON.stringify({
      model: 'MiniMax-M2.7',
      messages,
      temperature: 0.7,
    }),
  });

  const data = await response.json();
  console.log('MiniMax API response:', JSON.stringify(data, null, 2));

  if (!response.ok) {
    throw new Error(`MiniMax API error: ${response.statusText} - ${JSON.stringify(data)}`);
  }

  if (!data.choices || !data.choices[0]) {
    throw new Error(`Invalid API response: ${JSON.stringify(data)}`);
  }

  return data.choices[0].message.content;
}

// 去重函数
function deduplicateErrors(errors: LineError[]): LineError[] {
  const seen = new Set<string>();
  return errors.filter((err) => {
    const key = `${err.error.trim()}-${err.suggestion.trim()}`;
    if (seen.has(key)) return false;
    seen.add(key);
    return true;
  });
}

function deduplicateIssues(issues: Issue[]): Issue[] {
  const seen = new Set<string>();
  return issues.filter((iss) => {
    const key = `${iss.issue.trim()}-${iss.suggestion.trim()}`;
    if (seen.has(key)) return false;
    seen.add(key);
    return true;
  });
}

// 错误高亮组件 - 核心算法
function HighlightedContent({ content, errors }: { content: string; errors: LineError[] }) {
  const dedupedErrors = deduplicateErrors(errors);

  if (dedupedErrors.length === 0) {
    return <div className="whitespace-pre-wrap text-gray-800 leading-relaxed">{content}</div>;
  }

  // 查找所有错误位置
  interface ErrorPos { start: number; end: number; error: LineError; }
  const errorPositions: ErrorPos[] = [];

  dedupedErrors.forEach((err) => {
    const errorText = err.error.trim();
    if (!errorText) return;

    // 大小写不敏感搜索所有出现位置
    let searchStart = 0;
    let idx = content.toLowerCase().indexOf(errorText.toLowerCase(), searchStart);
    while (idx !== -1) {
      errorPositions.push({
        start: idx,
        end: idx + errorText.length,
        error: err,
      });
      searchStart = idx + 1;
      idx = content.toLowerCase().indexOf(errorText.toLowerCase(), searchStart);
    }
  });

  // 按位置排序
  errorPositions.sort((a, b) => a.start - b.start);

  // 去除重叠（保留第一个）
  const filtered: ErrorPos[] = [];
  let lastEnd = 0;
  errorPositions.forEach((pos) => {
    if (pos.start >= lastEnd) {
      filtered.push(pos);
      lastEnd = pos.end;
    }
  });

  if (filtered.length === 0) {
    return <div className="whitespace-pre-wrap text-gray-800 leading-relaxed">{content}</div>;
  }

  // 构建高亮内容
  const parts: React.ReactNode[] = [];
  let currentPos = 0;

  filtered.forEach((pos, idx) => {
    if (pos.start > currentPos) {
      parts.push(<span key={`text-${currentPos}`} className="text-gray-800">{content.slice(currentPos, pos.start)}</span>);
    }

    const highlightClass = getHighlightStyles(pos.error.type);
    parts.push(
      <span key={`error-${idx}`} className={`${highlightClass} px-1 rounded cursor-help`} title={`类型: ${pos.error.type}\n建议: ${pos.error.suggestion}`}>
        {content.slice(pos.start, pos.end)}
      </span>
    );
    currentPos = pos.end;
  });

  if (currentPos < content.length) {
    parts.push(<span key={`text-${currentPos}`} className="text-gray-800">{content.slice(currentPos)}</span>);
  }

  return <div className="whitespace-pre-wrap leading-relaxed">{parts}</div>;
}

### 5. 主应用 (client/src/App.tsx)
```typescript
import { useState } from 'react';
import type { GradingResult, EssayInput } from './types/essay';
import { GradingResult as GradingResultComponent } from './components/GradingResult';

function App() {
  const [input, setInput] = useState<EssayInput>({ topic: '', content: '' });
  const [result, setResult] = useState<GradingResult | null>(null);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState('');

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!input.topic || !input.content) return;

    setLoading(true);
    setError('');

    try {
      const response = await fetch('http://localhost:3001/api/grade', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(input),
      });

      if (!response.ok) throw new Error('批改请求失败');

      const data = await response.json();
      setResult(data);
    } catch (err) {
      setError('批改请求失败，请稍后重试');
      console.error(err);
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="min-h-screen bg-gray-100 py-8">
      <div className="max-w-4xl mx-auto px-4">
        <h1 className="text-3xl font-bold text-center mb-8 text-gray-800">IELTS 作文智能批改</h1>

        {!result ? (
          <form onSubmit={handleSubmit} className="bg-white rounded-lg shadow-md p-6 space-y-4">
            <div>
              <label className="block text-sm font-medium text-gray-700 mb-1">作文题目</label>
              <input
                type="text"
                value={input.topic}
                onChange={(e) => setInput({ ...input, topic: e.target.value })}
                className="w-full px-3 py-2 border border-gray-300 rounded-md focus:outline-none focus:ring-2 focus:ring-blue-500"
                placeholder="请输入作文题目"
              />
            </div>
            <div>
              <label className="block text-sm font-medium text-gray-700 mb-1">作文内容</label>
              <textarea
                value={input.content}
                onChange={(e) => setInput({ ...input, content: e.target.value })}
                rows={10}
                className="w-full px-3 py-2 border border-gray-300 rounded-md focus:outline-none focus:ring-2 focus:ring-blue-500"
                placeholder="请输入作文内容"
              />
            </div>
            {error && <p className="text-red-500 text-sm">{error}</p>}
            <button
              type="submit"
              disabled={loading}
              className="w-full bg-blue-500 text-white py-2 px-4 rounded-md hover:bg-blue-600 disabled:bg-gray-400"
            >
              {loading ? '批改中...' : '提交批改'}
            </button>
          </form>
        ) : (
          <div>
            <button
              onClick={() => setResult(null)}
              className="mb-4 text-blue-500 hover:underline"
            >
              ← 重新批改
            </button>
            <GradingResultComponent result={result} originalContent={input.content} />
          </div>
        )}
      </div>
    </div>
  );
}

export default App;
```

## 五、关键技术点

1. **AI Prompt 设计**：
   - 每个维度详细定义和例子
   - 严格要求输出 JSON 格式
   - 说明 error 必须是文章中实际存在的文本

2. **高亮算法**：
   - 大小写不敏感匹配
   - 全文搜索所有出现位置
   - 去除重叠部分

3. **前端状态管理**：
   - useState 管理当前选中的错误类型
   - 根据选中类型过滤显示的错误

4. **去重逻辑**：
   - 基于 error + suggestion 组合去重
   - 避免重复显示相同问题
